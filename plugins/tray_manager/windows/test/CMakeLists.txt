cmake_minimum_required(VERSION 3.15)
project(tray_manager_native_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  add_compile_options(/utf-8 /EHsc /UNDEBUG)
endif()
enable_testing()
add_executable(tray_integer_test tray_integer_test.cpp)
add_test(NAME tray_integer COMMAND tray_integer_test)
if(WIN32)
  add_executable(tray_menu_host_test tray_menu_host_test.cpp)
  target_link_libraries(tray_menu_host_test PRIVATE user32)
  add_test(NAME tray_menu_host COMMAND tray_menu_host_test)
  set_tests_properties(tray_menu_host PROPERTIES TIMEOUT 15)
endif()
